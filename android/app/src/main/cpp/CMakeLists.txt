cmake_minimum_required(VERSION 3.22.1)
project(farsicaption LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
if(NOT WHISPER_SOURCE_DIR)
    include(FetchContent)
    FetchContent_Declare(whisper_source
        URL https://github.com/ggml-org/whisper.cpp/archive/a8d002cfd879315632a579e73f0148d06959de36.tar.gz
        URL_HASH SHA256=7b17da903114ed45d82f48c030e3be6a5a8a279f884b2f733706dbb4e832fb6b)
    FetchContent_GetProperties(whisper_source)
    if(NOT whisper_source_POPULATED)
        FetchContent_Populate(whisper_source)
    endif()
    set(WHISPER_SOURCE_DIR ${whisper_source_SOURCE_DIR})
endif()
add_subdirectory(${WHISPER_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/whisper)
add_library(caption_core STATIC caption_core.cpp)
target_link_libraries(caption_core PUBLIC whisper)
target_include_directories(caption_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(ANDROID)
    add_library(farsicaption SHARED caption_jni.cpp)
    target_link_libraries(farsicaption PRIVATE caption_core)
    target_link_options(farsicaption PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
else()
    enable_testing()
    add_executable(native_tests native_tests.cpp)
    target_link_libraries(native_tests PRIVATE caption_core)
    add_test(NAME native_contract COMMAND native_tests)
    # Experimental decoder settings are allowed only in host benchmark builds.
    if(DEFINED CAPTION_BENCHMARK_BEAM_SIZE)
        target_compile_definitions(caption_core PRIVATE CAPTION_BENCHMARK_BEAM_SIZE=${CAPTION_BENCHMARK_BEAM_SIZE})
    endif()
    if(CAPTION_BENCHMARK_MERGE_VAD)
        target_compile_definitions(caption_core PRIVATE CAPTION_BENCHMARK_MERGE_VAD=1)
    endif()
    add_executable(asr_batch asr_batch.cpp)
    target_link_libraries(asr_batch PRIVATE caption_core)
    add_executable(asr_probe asr_probe.cpp)
    target_link_libraries(asr_probe PRIVATE caption_core)
    # Reproducible research conversion using this same pinned ggml implementation.
    add_executable(whisper_quantize EXCLUDE_FROM_ALL
        ${WHISPER_SOURCE_DIR}/examples/quantize/quantize.cpp
        ${WHISPER_SOURCE_DIR}/examples/common.cpp
        ${WHISPER_SOURCE_DIR}/examples/common-ggml.cpp)
    target_include_directories(whisper_quantize PRIVATE ${WHISPER_SOURCE_DIR}/examples)
    target_link_libraries(whisper_quantize PRIVATE whisper)
endif()
